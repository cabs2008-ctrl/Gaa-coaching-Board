function drawGoal(edgeX, isRight) { const h = canvas.height; const midY = h / 2; const bigDepth = 90; const smallDepth = 45; const goalCenterX = isRight ? edgeX : edgeX; const dir = isRight ? -1 : 1; // BIG RECTANGLE (20m area) ctx.strokeRect( goalCenterX, midY - 100, dir * bigDepth, 200 ); // SMALL RECTANGLE (goal box) ctx.strokeRect( goalCenterX, midY - 50, dir * smallDepth, 100 ); // SMALL D (correct — touches big box) ctx.beginPath(); ctx.arc( goalCenterX + dir * bigDepth, midY, 35, isRight ? -Math.PI/2 : Math.PI/2, isRight ? Math.PI/2 : 3*Math.PI/2 ); ctx.stroke(); // 40m ARC (correct — reaches near 20m line) ctx.beginPath(); ctx.arc( goalCenterX, midY, 140, isRight ? -Math.PI/2 : Math.PI/2, isRight ? Math.PI/2 : 3*Math.PI/2 ); ctx.stroke(); }
